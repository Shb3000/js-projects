    Exercise 8: Generator Function

    Create a generator function.

    function* countUp() {
  for (let i = 1; i <= 3; i++) {
    yield i;
  }
}

const gen = countUp();
console.log('First:', next(gen).value);
console.log('Second:', next(gen).value);
console.log('Third:', next(gen).value);

    Exercise 10: async Generator

    Create an async generator function.

    async function* asyncGen() {
  for (let i = 1; i <= 3; i++) {
    await new Promise(resolve => setTimeout(resolve, 100));
    yield i;
  }
}

const gen = asyncGen();
console.log('First:', next(gen).value);
console.log('Second:', next(gen).value);
console.log('Third:', next(gen).value);
