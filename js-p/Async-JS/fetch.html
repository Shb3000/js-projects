🌐 JavaScript Fetch API Exercise

Exercise 1: Basic Fetch GET Request

Make a simple GET request using fetch.

fetch('https://httpbin.org/get')
  .then(response => {
    console.log('Status:', response.status);
    console.log('OK:', response.ok);
  })
  .then(data => {
    console.log('Data received');
  })
  .catch(error => {
    console.log('Error:', error.message);
  });
Output:

Status: , OK: , Data received

Exercise 2: Fetch with JSON Response

Parse JSON response from a fetch request.

fetch('https://jsonplaceholder.typicode.com/posts/1')
  .then(response => {
    return response.json();
  })
  .then(data => {
    console.log('Title:', data.title);
    console.log('Body:', data.body);
  })
  .catch(error => {
    console.log('Error:', error.message);
  });
Output:

Title: , Body length: chars

Exercise 4: POST Request with Body

Send data in a POST request.

fetch('https://jsonplaceholder.typicode.com/posts', {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json'
  },
  body: JSON.stringify({
    title: 'New Post',
    body: 'This is a new post.',
    userId: 1
  })
})
  .then(response => response.json())
  .then(data => {
    console.log('Post created:', data.title);
  })
  .catch(error => {
    console.log('Error:', error.message);
  });
Output:

Post created:

Exercise 7: Fetch with Timeout

Handle slow responses with a timeout.

const controller = new AbortController();
const timeoutId = setTimeout(() => controller.abort(), 2000);

fetch('https://jsonplaceholder.typicode.com/posts/1', {
  signal: controller.signal
})
  .then(response => response.json())
  .then(data => {
    console.log('Data received:', data.title);
    clearTimeout(timeoutId);
  })
  .catch(error => {
    if (error.name === 'AbortError') {
      console.log('Request timed out');
    } else {
      console.log('Error:', error.message);
    }
  });
Output:

Request timed out

Exercise 8: Fetch Chain with Multiple Resources

Chain fetch requests for multiple resources.

const user = await fetch('https://jsonplaceholder.typicode.com/users/1')
  .then(response => response.json());

const post = await fetch('https://jsonplaceholder.typicode.com/posts/1')
  .then(response => response.json());

console.log('User:', user.name);
console.log('Post:', post.title);
Output:

User: , Post:

Exercise 9: Fetch with Error Handling

Handle network errors and invalid responses.

try {
  const response = await fetch('https://httpbin.org/status/404');
  
  if (!response.ok) {
    throw new Error('Request failed with status ' + response.status);
  }
  
  const data = await response.json();
  console.log('Status:', data.status);
} catch (error) {
  console.log('Error:', error.message);
}
Output:

Status:

Exercise 10: Fetch with Caching

Use cache-related headers with fetch.

fetch('https://httpbin.org/get', {
  cache: 'reload',
  credentials: 'include'
})
  .then(response => {
    console.log('Cache mode:', response.headers.get('Cache-Control'));
    console.log('Credentials:', response.headers.get('WWW-Authenticate'));
  })
  .catch(error => {
    console.log('Error:', error.message);
  });
Output:

Cache mode: , Credentials:
