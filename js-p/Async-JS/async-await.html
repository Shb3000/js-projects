Exercise 2: async with setTimeout

Use async/await with setTimeout to delay execution.

async function delayedMessage(delay) {
  await new Promise(resolve => setTimeout(resolve, delay));
  return 'Delayed: ' + delay + 'ms';
}

const result = await delayedMessage(1000);
console.log('Result:', result);

Exercise 4: await in Loop

Use await inside a for loop.

async function processNumbers() {
  const numbers = [1, 2, 3, 4, 5];
  const result = [];
  
  for (let num of numbers) {
    await new Promise(resolve => setTimeout(resolve, 100));
    result.push(num * 2);
  }
  
  return result;
}

const result = await processNumbers();
console.log('Doubled:', result);
Output: Doubled =

Exercise 9: Nested async Functions

Use nested async functions with await.

async function outer() {
  console.log('Outer started');
  const inner = async function() {
    await new Promise(resolve => setTimeout(resolve, 100));
    console.log('Inner completed');
  };
  const result = await inner();
  console.log('Outer completed');
  return result;
}

const result = await outer();
console.log('Final result:', result);
