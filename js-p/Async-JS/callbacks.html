📞 JavaScript Callbacks Exercise

Exercise 1: Basic Callback Function

Pass a function as an argument to another function.

function greet(name) {
  return 'Hello, ' + name + '!';
}

function sendGreeting(greetingFunction) {
  const name = 'Alice';
  const message = greetingFunction(name);
  console.log(message);
}

sendGreeting(greet);
Output:

Hello, Alice!

Exercise 2: Callback with Arguments

function multiply(a, b) {
  return a * b;
}

function addNumbers(a, b) {
  return multiply(a, b);
}

const result = addNumbers(5, 3);
console.log('Result:', result);

Exercise 3: Nested Callbacks

Use callbacks within callbacks.

function process(data) {
  return function(callback) {
    return callback(data);
  };
}

function calculate(value) {
  return value * 2;
}

const result = process(5)(calculate);
console.log('Result:', result);
Output: Result =

Exercise 4: Callback with Delay

Use setTimeout as a callback.

function delayedCallback(callback, delay) {
  setTimeout(callback, delay);
}

delayedCallback(() => {
  console.log('Delayed message');
}, 1000);
Output: Delayed message

Exercise 5: Callback with Error Handling

Handle errors in callback functions.

function safeCallback(callback) {
  try {
    callback();
  } catch (error) {
    console.log('Error caught:', error);
  }
}

safeCallback(() => {
  console.log('Success!');
});
Output: Success!

Exercise 6: Callback with Multiple Parameters

Pass a function with multiple parameters to another function.

function logValues(a, b, c) {
  return function(callback) {
    callback(a, b, c);
  };
}

const result = logValues(1, 2, 3)((x, y, z) => {
  console.log('x=' + x + ', y=' + y + ', z=' + z);
});

result();
Output: x=1, y=2, z=3

Exercise 7: Callback Pattern (Callback Hell)

Identify the callback hell pattern (nested callbacks).

function step1() {
  return function(callback) {
    setTimeout(() => {
      console.log('Step 1 completed');
      callback(function(callback2) {
        setTimeout(() => {
          console.log('Step 2 completed');
          callback2(function(callback3) {
            setTimeout(() => {
              console.log('Step 3 completed');
              callback3();
            }, 100);
          });
        }, 100);
      });
    }, 100);
  };
}

    Exercise 10: Callback with Multiple Returns

    Return multiple values through callbacks.

    function operation(a, b, callback) {
  const sum = a + b;
  const product = a * b;
  callback(sum, product);
}

const result = operation(2, 3, function(s, p) {
