  isStudent: true

Exercise 4: parse with Reviver Function

Parse a JSON string using a reviver function to transform data.

const data = '{"name":"John","age":30,"skills":["JS","React"]}';
const parsed = JSON.parse(data, (key, value) => {
  if (typeof value === 'string') {
    return value.toUpperCase();
  }
  return value;
});
Output: name = , age = , skills =

Exercise 5: Stringify with Replacer

Stringify an object using a replacer function to exclude certain properties.

const user = {
  name: 'Alice',
  age: 25,
  address: { city: 'NYC', zip: '10001' }
};
const json = JSON.stringify(user, (key, value) => {
  return key !== 'address';
});
Output: JSON (no address) =

Exercise 6: parse with Reviver (Number Conversion)

Use a reviver function to convert string numbers to actual numbers.

const data = '{"price":"100","quantity":"5"}';
const parsed = JSON.parse(data, function(key, value) {
  if (typeof value === 'string' && !isNaN(value)) {
    return parseFloat(value);
  }
  return value;
});

Exercise 7: Circular Reference Handling

};
circular.self = circular;

Exercise 10: Parse JSON with Reviver (Custom Objects)

Use a reviver to convert string keys to numbers and values to specific types.

const data = '{"id":"100","name":"Bob","age":"25"}';

  return {
    [key]: value === 'true' ? true : value === 'false' ? false : 
           value === 'null' ? null :

  };
});
